  p_values_row <- gene_wise_p[i, ]
  
  # Initialize an empty list to store Bonferroni p-values and module names for each module
  module_acat_pvals <- c()
  module_names <- c()

  # Iterate over each module in gene_modules to combine p-values using Bonferroni
  for (j in seq_along(gene_modules)) {
    module_name <- names(gene_modules)[j]  # Get the module name
    module_genes <- gene_modules[[j]][gene_modules[[j]] %in% colnames(p_values_row)]
    
    # Only apply Bonferroni if there are genes to combine and they are not NA
    if (length(module_genes) > 0) {
      module_p_values <- as.numeric(p_values_row[module_genes])
      module_p_values <- module_p_values[!is.na(module_p_values)]  # Remove NAs
      
      # Check if module_p_values still has values after removing NAs
      if (length(module_p_values) > 0) {
        # Bonferroni adjustment: multiply smallest p-value by number of tests
        bonferroni_pval <- min(module_p_values) * length(module_p_values)
        bonferroni_pval <- min(bonferroni_pval, 1)  # Cap p-values at 1
        module_acat_pvals <- c(module_acat_pvals, bonferroni_pval)
        module_names <- c(module_names, module_name)
      } else {
        module_acat_pvals <- c(module_acat_pvals, NA)

      }
    } else {
