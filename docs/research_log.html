t_test_wilcoxon <- function(response_matrix, grna_assignment_matrix) {


  response_matrix <- t(apply(as.matrix(response_matrix), 1, rank, ties.method = "average"))

  # Step 3: Subtract row-wise minimum
  min_vals <- apply(response_matrix, 1, min)
  response_matrix <- response_matrix - min_vals

  response_matrix <- as(response_matrix, "dgCMatrix")

  if (!inherits(grna_assignment_matrix, "dgCMatrix")) {
    grna_assignment_matrix <- as(as.matrix(grna_assignment_matrix), "dgCMatrix")
  }

  n_cells <- ncol(response_matrix)
  n_genes <- nrow(response_matrix)
  grna_infected_cell_count <- rowSums(grna_assignment_matrix)

  mean_expression_trt <- tcrossprod(grna_assignment_matrix, response_matrix) / grna_infected_cell_count

  grna_control_cell_count <- n_cells - grna_infected_cell_count


  total_expression <- Matrix::rowSums(response_matrix)

  mean_expression_control <- t(total_expression - t(mean_expression_trt * grna_infected_cell_count)) / grna_control_cell_count


  mean_expression <- total_expression/ncol(response_matrix)
  mean_expression_squared <- rowSums(response_matrix^2)/ncol(response_matrix)


  var_expression <- mean_expression_squared-mean_expression^2


  t_statistics <- t(t((mean_expression_trt - mean_expression_control)/sqrt(1/grna_infected_cell_count+1/grna_control_cell_count)*n_cells/(n_cells-1)) / sqrt(var_expression))

  return(t_statistics)
}

This will result in a situation in resampling where no cells receiving the perturbation express the genes, leading to an extreme null z value.

This can be proved by the another finding that the presence of abnormal null distribution is strongly correlated with the number of cells express the gene.
